    ```python
                            file_name: str, password: str = "") -> bytes:
        """
        FIRMWARE_UPLOAD_INIT(0xD1) 패킷 — 86바이트 (FW-02).

        payload 79B 레이아웃:
          offset  0~ 5: dst_mac[6]
          offset  6:    pw_len[1]
          offset  7~22: pw[16]

        """
        mac_b = _mac_str_to_bytes(target_mac)                                     # 6B
        pw_bytes = password.encode('ascii', errors='replace')[:16]
        pw_enc = pw_bytes.ljust(16, b'\x00')                                       # 16B
    ```

    ```python
    def _is_fw_done_reply(data: bytes) -> bool:
        """0xD2 응답 확인: data[0]==0xA5, data[3]==0xD2, data[4]==0x55"""
        return (len(data) >= 7
    ```

    ```python
    def _run_auto(self):
        import shutil
        tmpdir = tempfile.mkdtemp(prefix='wiz550fw_')
        fw_filename = os.path.basename(self.fw_path)
        dst = os.path.join(tmpdir, fw_filename)
        shutil.copy2(self.fw_path, dst)

        try:
            self._server = tftpy.TftpServer(tftproot=tmpdir)
        except Exception as e:
            self.error.emit(f"TFTP 서버 초기화 실패: {e}")
            self.finished.emit(False)
            shutil.rmtree(tmpdir, ignore_errors=True)
            return

        listen_thread = threading.Thread(
            target=self._server.listen,
            args=(self.server_ip, self.server_port),
            daemon=True,
            name="tftpy-listen",
        )
        try:
            listen_thread.start()
        except Exception as e:
            self.error.emit(str(e))
            self.finished.emit(False)
            shutil.rmtree(tmpdir, ignore_errors=True)
            return

        # 서버 시작 대기 — OSError(포트 바인딩 실패)는 listen() 내부에서 발생
        # is_running.wait()으로 직접 확인 불가(listen 내부 try/except가 삼킬 수 있음)
        # 대신 threading.Event로 start 확인 + listen_thread.is_alive() 병행
        if not self._server.is_running.wait(timeout=3.0):
            # 서버 시작 안 됨
            if not listen_thread.is_alive():
                self.error.emit(
                    "Port 69 requires Administrator privileges. "
                    "Please use the manual tab with an external TFTP server."
                )
            else:
                self.error.emit("TFTP 서버 시작 타임아웃")
            self.finished.emit(False)
            shutil.rmtree(tmpdir, ignore_errors=True)
            return

        self.progress.emit("Sending firmware upload request...")
        # _wait_d2가 내부적으로 소켓 생성·bind 후 _send_fw_init() 호출하고 0xD2 대기
        self.progress.emit("Waiting for device response...")
        success = self._wait_d2()

        self._server.stop(now=True)
        listen_thread.join(timeout=3)
        shutil.rmtree(tmpdir, ignore_errors=True)

        self.finished.emit(success)
    ```

    **소켓 전략 (Java 원본 분석으로 확정 — 2026-05-20):**
    Java DatagramSocket()는 랜덤 에페머럴 포트에 바인드하고 동일 소켓으로 전송+수신.
    0xD1 Init Reply(op=0xD1, reply=0x55)와 0xD2 Done Reply 모두 소스포트(에페머럴)로 응답.
    → _send_fw_init()와 _wait_d2()가 self._d2_sock (단일 소켓) 을 공유.
    → port 6550 고정 bind 불필요. 에페머럴 bind(0)가 기본 전략.

    **_wait_d2 구현 (단일 소켓 — 에페머럴 bind 후 전송+수신):**

    Java 원본: `new DatagramSocket()` (랜덤 포트) → 같은 소켓으로 전송·수신.
    장치 응답(0xD1 Init Reply + 0xD2 Done Reply)은 0xD1 패킷의 소스 포트로 옴.
    bind(0) 실패는 OS 자원 고갈 등 극히 예외적 상황 — 예외처리는 유지.

    ```python
    def _wait_d2(self, timeout_sec: float = 30.0) -> bool:
        # 단일 소켓 생성 + 에페머럴 바인드 (포트 0 → OS 배정)
        self._d2_sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        self._d2_sock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        try:
            self._d2_sock.bind(('', 0))
        except OSError as e:
            self._d2_sock.close()
            self._d2_sock = None
            self.error.emit(
                f"Socket bind failed entirely — cannot receive device response: {e}"
            )
            return False

        # 0xD1 전송 (같은 소켓 — 소스 포트가 응답 수신 포트)
        try:
            self._send_fw_init()
        except Exception as e:
            self._d2_sock.close()
            self._d2_sock = None
            raise  # 호출자(_run_auto/_run_manual)에서 처리

        self._d2_sock.setblocking(False)
        deadline = time.time() + timeout_sec
        try:
            while not self._stop_event.is_set():
                remaining = deadline - time.time()
                if remaining <= 0:
                    return False
                ready, _, _ = select.select([self._d2_sock], [], [], min(remaining, 1.0))
                if not ready:
                    continue
                data, _ = self._d2_sock.recvfrom(256)
                if _is_fw_done_reply(data):
                    return True
        finally:
            self._d2_sock.close()
            self._d2_sock = None
        return False
    ```
